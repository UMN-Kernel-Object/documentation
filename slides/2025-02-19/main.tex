\begin{frame}{Environment}
	\begin{itemize}
		\item If you rebooted, or for whatever other reason \texttt{lfs.img} got unmounted from \texttt{\$LFS}:
		      \begin{itemize}
			      \item \texttt{sudo losetup -{}-partscan -{}-find -{}-show lfs.img}
			      \item \texttt{sudo mount /dev/loop0 \$LFS}
		      \end{itemize}
		      \vspace{0.5em}
		\item Check that these environment variables are set:
		      \begin{itemize}
			      \item \texttt{\$LFS}: Should be the directory you were working in before
			      \item \texttt{\$LFS\_TGT}: \texttt{x86\_64-lfs-linux-gnu}
			      \item \texttt{\$MAKEFLAGS}: Something like \texttt{-j16 -l16}
			      \item \texttt{\$PATH}: The first thing should be \texttt{\$LFS/tools/bin}
			      \item Consider putting these in \texttt{\textasciitilde/.bashrc}
		      \end{itemize}
		      \vspace{0.5em}
		\item Check that these packages are installed:
		      \begin{itemize}
			      \item \texttt{build-essential}, \texttt{bison}
			      \item \texttt{libgmp-dev}, \texttt{libmpc-dev}, \texttt{libmpfr-dev}
			      \item \texttt{texinfo}
		      \end{itemize}
	\end{itemize}
\end{frame}
